export function formatDate(iso: string | null | undefined, opts?: Intl.DateTimeFormatOptions): string {
  if (!iso) return "";
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return "";
  return d.toLocaleDateString("en-US", opts ?? { month: "short", day: "numeric", year: "numeric" });
}

export function formatNumber(n: number): string {
  return new Intl.NumberFormat("en-US").format(n);

export function LimitReached({ usage }: { usage: UsageSummary }) {
  const reset = formatDate(usage.resetAt, { month: "long", day: "numeric" });
  let title: string;
  let body: string;
  let primary: { href: string; label: string };
  const secondary: { href: string; label: string } | null =
    usage.plan === "trial" ? null : { href: "/account", label: "View usage" };

  if (usage.plan === "trial") {
    title = "Your trial replies are used up";
    body = `You've used all ${usage.limit} trial ${usage.unitLabel}. Pro and Max run the same Citad3L V1 engine with a monthly allowance. Failed replies never count.`;
    primary = { href: "/pricing", label: "See plans" };
  } else if (usage.plan === "pro") {
    title = "You've reached this month's allowance";
    body = `Your Pro allowance${reset ? ` resets on ${reset}` : " resets on your billing date"}. Max gives you 2.5× the allowance on the same engine.`;
    primary = { href: "/pricing", label: "Upgrade to Max" };
  } else {
    title = "You've reached this month's allowance";
    body = `Your Max allowance${reset ? ` resets on ${reset}` : " resets on your billing date"}. Failed or stopped replies are never charged.`;
    primary = { href: "/account", label: "Manage plan" };
  }

  return (
    <div className="limit-card" role="alert">
      <div className="limit-icon"><LockIcon /></div>
      <div>
        <div className="limit-title">{title}</div>
        <p className="limit-body">{body}</p>
        <div className="limit-actions">
          <Link href={primary.href} className="btn btn-primary btn-sm">{primary.label}</Link>
          {secondary && primary.href !== secondary.href ? (
            <Link href={secondary.href} className="btn btn-secondary btn-sm">{secondary.label}</Link>
          ) : null}
        </div>
      </div>
    </div>
  );
}

interface NavItem { href: string; label: string; match: (p: string) => boolean; }

const LINKS: NavItem[] = [
  { href: "/", label: "Explore", match: (p) => p === "/" },
  { href: "/chat", label: "Chat", match: (p) => p === "/chat" || p.startsWith("/chat/") },
  { href: "/models", label: "Models", match: (p) => p === "/models" || p.startsWith("/models/") },

function initials(label?: string): string {
  if (!label) return "C";
  const base = label.includes("@") ? label.split("@")[0] : label;
  const parts = base.split(/[\s._-]+/).filter(Boolean);
  const s = parts.length > 1 ? parts[0][0] + parts[1][0] : base.slice(0, 2);
  return s.toUpperCase() || "C";
}

export interface TopNavProps { signedIn: boolean; userLabel?: string; }

export function TopNav({ signedIn, userLabel }: TopNavProps) {
  const pathname = usePathname() ?? "/";
  const [open, setOpen] = React.useState(false);
  const burgerRef = React.useRef<HTMLButtonElement>(null);
  const sheetRef = React.useRef<HTMLDivElement>(null);
  const [isPhone, setIsPhone] = React.useState(false);

  React.useEffect(() => {
    const mq = window.matchMedia("(max-width: 860px)");
    const update = () => {
      setIsPhone(mq.matches);
      if (!mq.matches) setOpen(false);
    };
    update();
    mq.addEventListener("change", update);
    return () => mq.removeEventListener("change", update);
  }, []);

  React.useEffect(() => { setOpen(false); }, [pathname]);

  React.useEffect(() => {
    if (!open) return;
    const onKey = (e: KeyboardEvent) => {
      if (e.key === "Escape") {
        setOpen(false);
        burgerRef.current?.focus();
      }
    };
    document.addEventListener("keydown", onKey);
    const prev = document.body.style.overflow;
    document.body.style.overflow = "hidden";
    sheetRef.current?.querySelector<HTMLElement>("a,button")?.focus();
    return () => {
      document.removeEventListener("keydown", onKey);
      document.body.style.overflow = prev;
    };
  }, [open]);

  const pricingActive = pathname.startsWith("/pricing");
  const accountActive = pathname.startsWith("/account");

  return (
    <header className="nav">
      <div className="nav-inner">
        <Link href="/" className="nav-brand" aria-label="Citad3L home">
          <Rook size={isPhone ? 22 : 18} />
          <span>Citad3L</span>
        </Link>

        {!isPhone && (
          <nav className="nav-links" aria-label="Primary">
            {LINKS.map((l) => (
              <Link key={l.href} href={l.href} className="nav-link" aria-current={l.match(pathname) ? "page" : undefined}>
                {l.label}
              </Link>
            ))}
          </nav>
        )}

        <div className="nav-right">
          <Link href="/chat" className="icon-btn nav-icon" aria-label="Search chats">
            <SearchIcon />
          </Link>
          {!isPhone ? (
            <div className="nav-desktop">
              <Link href="/projects" className="icon-btn nav-icon" aria-label="Projects">
                <FolderIcon />
              </Link>
              <Link href="/pricing" className="nav-text" aria-current={pricingActive ? "page" : undefined}>
                Pricing
              </Link>
              {signedIn ? (
                <Link
                  href="/account"
                  className="nav-avatar"
                  aria-label={`Account${userLabel ? ` (${userLabel})` : ""}`}
                  aria-current={accountActive ? "page" : undefined}
                  title={userLabel}
                >
                  {initials(userLabel)}
                </Link>
              ) : (
                <>
                  <Link href="/login" className="btn btn-secondary btn-sm">Log in</Link>
                  <Link href="/login?next=/chat" className="btn btn-primary btn-sm">Get started</Link>
                </>
              )}
            </div>
          ) : (
            <button
              ref={burgerRef}
              type="button"
              className="icon-btn nav-icon"
              style={{ display: "inline-grid" }}
              aria-label={open ? "Close menu" : "Open menu"}
              aria-expanded={open}
              aria-controls="nav-sheet"
              onClick={() => setOpen((o) => !o)}
            >
              <MenuIcon open={open} />
            </button>
          )}
        </div>
      </div>

      {isPhone && open && (
        <>
          <div className="sheet-backdrop" onClick={() => setOpen(false)} aria-hidden="true" />
          <div id="nav-sheet" ref={sheetRef} className="sheet" role="dialog" aria-modal="true" aria-label="Menu">
            {LINKS.map((l) => (
              <Link key={l.href} href={l.href} className="sheet-link" aria-current={l.match(pathname) ? "page" : undefined}>

              </Link>
            ))}
